import ezdxf


def print_dxf(path):

    doc = ezdxf.readfile(str(path))

    msp = doc.modelspace()
    for insert in msp.query("INSERT"):
        for attrib in getattr(insert, "attribs", []):
            tag = str(getattr(attrib.dxf, "tag", "") or "").upper()
            text = str(getattr(attrib.dxf, "text", "") or "").upper()
            if text == "F21-1-1GPB-N3-01-20Q22":
                print(f"tag:{tag}:{text}")
